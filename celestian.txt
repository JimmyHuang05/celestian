
server {
    # 监听 80 端口 (HTTP)
    listen 80;
    listen [::]:80;

    # 你的子域名
    server_name celestian.jimmyhuang.cn;

    # 网页文件存放的根目录
    root /var/www/celestian;

    # 默认首页文件名
    index index.html index.htm;

    # 访问日志和错误日志（建议加上，方便以后排查问题）
    access_log /var/log/nginx/celestian.access.log;
    error_log /var/log/nginx/celestian.error.log;

    # ========== 新增的配置 ==========
    
    # 1. 前端 SPA — 所有页面请求指向 index.html
    location / {
        try_files $uri $uri/ /index.html;
    }

    # 2. 静态资源缓存（js/css/图片/字体）
    location /assets/ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }

    # 3. 数据文件缓存
    location /data/ {
        expires 30d;
        add_header Cache-Control "public";
    }

    # 4. HTML 编辑器工具（直接提供静态 HTML）
    location /editor/ {
        index index.html;
        try_files $uri $uri/ =404;
    }

    # 5. 百度验证
    location = /baidu_verify_codeva-9nWPZdui8T.html {
        try_files $uri =404;
    }

    # 禁止访问隐藏文件（如 .git）
    location ~ /\. {
        deny all;
    }
}
